state0 reads cellA (value-decode -> meaning), converges by target to a hub:
 - meaning->s12 hub: 'i' -> state12's decoder block (fixed entry d)

state12 reads cellB, converges to print0/print1. Uses the state6_fix fixes
(constant setup after _build_constants; low-L cells keep nop char)."""

INSTR={6,7,29,35,48,65,66,84}
def rot(v): return v//3+(v%3)*_ROTATE
def admissible(a): return sorted(33+((i-a)%94) for i in INSTR)
PART=[{0,2,4,6,9,11,14,26,51,72,84,91,93},{7,16,25,30,37,40,49,54,60,63,67,70,81},

def meaning(T,par):
    v=(T-33-par)%94
    for m,S in enumerate(PART):
        if v in S: return m
KA={'K0':0,'K1':29524,'K2':59048}
KVA,KVB=242,728
def LANDK(K,T): return _crazy(K,T)+1
def KINV(K): return (K%19683)*3+K//19683
KOF={}
def bfs(start,depth=11):
    seen={start:[]};fr=[start]
    for _ in range(depth):
        nf=[]
        for v in fr:
            for o in ('rot','K0','K1','K2'):
                nv=_rot(v) if o=='rot' else _crazy(KA[o],v)
                if nv not in seen: seen[nv]=seen[v]+[o];nf.append(nv)
        fr=nf
    return seen
# best_shared13 state 0 (cell 1) transitions per meaning; state 12 (cell 2) terminal
S0=[-2,12,12,10,9,-2,-1]      # meaning-> next(state) or -1/-2 (print0/print1)
S12=[-2,-1,-1,-2,-1,-1,-1]
# For a clean 2-state test, route s9,s10 (unbuilt) same as s12 (they won't be exercised).
def tgt0(m):
    t=S0[m]
    if t in (9,10,12): return 's12'
    return 'p0' if t==-1 else 'p1'
def tgt12(m):
    return 'p0' if S12[m]==-1 else 'p1'

def emit_read_converge(P,zc,Tcell,helper,used,data,target_hub):
    """Emit: read Tcell (value-decode) -> converge to target's hub -> i.
    target_hub: dict target-label -> (H, Pptr). Trampolines route each char's L to
    its meaning's target hub. Returns nothing; appends to P."""
    par=Tcell%2
    # trampolines per char -> H(target)
    trcells={}
    for Tp in admissible(Tcell):
        m=meaning(Tp,par); lab=target_of(Tcell,m); H=target_hub[lab][0]; L=LANDK(KOF[Tcell],Tp)
        placed=None
        cands=[_char_for('o',L)] if L<_ENTRY else _valid_chars(L)
        for cL in cands:
            tr=(_g(L) if L<_ENTRY else cL)+1
            if 34<=tr<_ENTRY and tr not in used and tr not in trcells:
                r=bfs(_g(tr))
                if H in r: placed=(cL,tr,r[H]); break
        assert placed,(Tp,'no tr',lab,H); cL,tr,ch=placed; used.add(tr); trcells[tr]=ch
        if L>=_ENTRY: data[L]=cL
    for tr,ch in trcells.items(): _emit_chain(P,tr,' '.join(ch),helper)

target_of=None  # set per state below

def build(TA,TB,cfg):
    (zcA,TcA,zcB,TcB,C0,C0v,C1,C1v,
     H_s12,P_s12, H0a,P0a,H1a,P1a, H0b,P0b,H1b,P1b, blkB, KcA,KcB)=cfg
    used={128,129}; mem={a:_g(a) for a in range(_ENTRY)}
    helper={'all1':128,'all2':129}
    for n,v in _T_HELPERS.items():
        a=next(x for x in range(130,_ENTRY) if x not in used and _g(x)==v); used.add(a); helper[n]=a
    for c in (zcA,zcB,C0,C1,P_s12,P0a,P1a,P0b,P1b,KcA,KcB): used.add(c)
    KOF[TcA]=KVA; KOF[TcB]=KVB
    data={TcA:TA,TcB:TB}
    P=_Planner(_ENTRY+1,34+(7-_ENTRY)%94,mem,data); P.code[_ENTRY]='j'
    _build_constants(P,helper)
    for c in (helper['all1'],helper['all2']): P.op('p',c); P.op('p',c)
    P.op('*',helper['w']); P.op('p',helper['all2'])
    # constants + hub pointer cells (P_* hold hub values H_*)
    chains=[(zcA,TcA-1),(zcB,TcB-1),(C0,C0v),(C1,C1v),
            (P_s12,H_s12),(P0a,H0a),(P1a,H1a),(P0b,H0b),(P1b,H1b),(KcA,KINV(KVA)),(KcB,KINV(KVB))]
    for cell,val in chains:
        r=bfs(_g(cell)); assert val in r,(cell,val); _emit_chain(P,cell,' '.join(r[val]),helper)
    # hub cells rotate to name their target block/stub
    def hub_stub(Hp1,avoid,floor=44000):
        for hc in _valid_chars(Hp1):
            for k,s in _rotations(hc):
                if s+1>=floor and s+1+8<_WORDS and not any((s+1+o) in avoid for o in range(4)):
                    return hc,k,s+1
        raise RuntimeError('no rot '+str(Hp1))
    avoid={x for H in (H_s12,H0a,H1a,H0b,H1b) for x in (H,H+1,H+2)}
    # state12 block entry is a chosen address blkB; the s12 hub must name blkB.
    # Find hc,k for H_s12 s.t. rot^k(hc)+1 == blkB -> instead place blkB where rotation lands.
    def hub_to(Hp1,want=None):
        for hc in _valid_chars(Hp1):
            for k,s in _rotations(hc):
                if want is not None:
                    if s+1==want and not any((s+1+o) in avoid for o in range(4)): return hc,k,s+1
                else:
                    if s+1>=44000 and s+1+8<_WORDS and not any((s+1+o) in avoid for o in range(6)): return hc,k,s+1
        return None
    # each print hub names its OWN stub; s12 hub names blkB
    hc0a,k0a,stub0a=hub_to(H0a+1); [avoid.add(stub0a+o) for o in range(6)]
    hc1a,k1a,stub1a=hub_to(H1a+1); [avoid.add(stub1a+o) for o in range(6)]
    hc0b,k0b,stub0b=hub_to(H0b+1); [avoid.add(stub0b+o) for o in range(6)]
    hc1b,k1b,stub1b=hub_to(H1b+1); [avoid.add(stub1b+o) for o in range(6)]
    hcs,ks,blkB=hub_to(H_s12+1); [avoid.add(blkB+o) for o in range(40)]
    for cell,hc in ((H0a+1,hc0a),(H1a+1,hc1a),(H0b+1,hc0b),(H1b+1,hc1b),(H_s12+1,hcs)):
        data[cell]=hc
    # trampolines for state A and B
    global target_of
    target_hubA={'s12':(H_s12,P_s12),'p0':(H0a,P0a),'p1':(H1a,P1a)}
    target_hubB={'p0':(H0b,P0b),'p1':(H1b,P1b)}
    target_of=lambda Tc,m: tgt0(m) if Tc==TcA else tgt12(m)
    emit_read_converge(P,zcA,TcA,helper,used,data,target_hubA)
    emit_read_converge(P,zcB,TcB,helper,used,data,target_hubB)
    # rotate all hub cells
    for Pp,H,k in ((P_s12,H_s12,ks),(P0a,H0a,k0a),(P1a,H1a,k1a),(P0b,H0b,k0b),(P1b,H1b,k1b)):
        for _ in range(k):
            P.hub(Pp,H); P.raw('*')
        P.d=H+2
    # resync model
    noneset={a for a in range(_ENTRY) if P.mem.get(a) is None}
    P.mem={a:_g(a) for a in range(_ENTRY)}
    for a in noneset: P.mem[a]=None
    # ---- state A decoder (inline) ----
    def decoder(zc,Tcell):
        P.op('*',KcA)
        P.goto(zc); P.raw('j'); P.d=Tcell; P.raw('p'); P.d=Tcell+1
        P.goto(zc); P.raw('j'); P.d=Tcell; P.raw('j')     # d=L
        P.raw('j'); P.raw('j'); P.raw('i')                # tr->hub(d=H+1)->i
    decoder(zcA,TcA)
    # ---- state B decoder block at blkB (entered by i from s12 hub, d=H_s12+2) ----
    QB=_Planner(blkB,H_s12+2,mem,data)
    QB.op('*',KcB)
    QB.goto(zcB); QB.raw('j'); QB.d=TcB; QB.raw('p'); QB.d=TcB+1
    QB.goto(zcB); QB.raw('j'); QB.d=TcB; QB.raw('j')
    QB.raw('j'); QB.raw('j'); QB.raw('i')
    for a,op in QB.code.items(): P.code[a]=op
    # per-hub print stubs (goto Ca,*,<,v), entry d=H+2.
    for stub,Ca,Hentry in ((stub0a,C0,H0a),(stub1a,C1,H1a),(stub0b,C0,H0b),(stub1b,C1,H1b)):
        Q=_Planner(stub,Hentry+2,dict(P.mem),data); Q.op('*',Ca); Q.raw('<'); Q.raw('v')
        for a,op in Q.code.items(): P.code[a]=op
    src=[_char_for('o',a) for a in range(_WORDS)]
    for a,op in P.code.items(): src[a]=_char_for(op,a)
    for a,ch in data.items(): src[a]=ch
    return ''.join(chr(x) for x in src)

def trace(src,steps=1000000):
    mem=list(_initial_memory(src)); st=(0,0,0,False); out=[]
    for _ in range(steps):
        st,wr,eff=_advance(st,mem)
        for a,v in wr: mem[a]=v
        if eff is not None: out.append(eff)
        if st[3]: break
    return st,out
