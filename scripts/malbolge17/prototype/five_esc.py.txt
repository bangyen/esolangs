"""Five-state shared decoder on one 3-cell group, REAL source.

States come from scripts/malbolge17/decoder_s5_norepeat.p.  State q reads cell
pos[q][s] of the group under view constant K[q][parity], lands at crazy(K,T)+1,
and routes through a trampoline to one of four SHARED hubs: print0, print1,
state 3's block, state 4's block.  One program per row s; every meaning
triple x in 7^3 is then run by patching the three table characters.
"""
import sys, itertools, subprocess
from pathlib import Path
HERE=Path(__file__).resolve().parent
exec((HERE/'chain_k.py.txt').read_text().split("def emit_read_converge")[0])

KS=[242,728,1700,2186,4616,5102,6074,6560,13364,13850]   # view constants, disjoint landings
BASES=[364,1093,2551,3280,6925]
PARITY=[58318,59047]
MASK=729
def load_code(fn):
    L=[l for l in open(fn).read().splitlines() if l.strip()]
    start=[int(x) for x in L[1].split()[1:]]; D=[];Pos=[]
    for q in range(5):
        a,b=L[2+q].split('| pos'); D.append([int(x) for x in a.split('|')[1].split()]); Pos.append([int(x) for x in b.split()])
    return start,D,Pos
START,DELTA,POS=load_code(str(HERE.parent/'decoder_s5_norepeat.p'))
TGT={-1:'p0',-2:'p1',3:'s3',4:'s4'}

def expected(s,x):
    q=START[s]
    while True:
        t=DELTA[q][x[POS[q][s]]]
        if t<0: return '0' if t==-1 else '1'
        q=t

from esolangs.tools.malbolge import _T_LABELS
LABELS={t+34:lab for t,lab in enumerate(_T_LABELS) if lab in "01xn"}
OTHERS=sorted(set(range(34,128))-set(LABELS))
INSTR8=[6,7,29,35,48,65,66,84]
def admits(addr,ch): return (ch-33+addr)%94 in INSTR8
KEY={'0':'p0','1':'p1','x':'s3','n':'s4'}; LABOF={v:k for k,v in KEY.items()}
def f1(v): return _crazy(v,29524)
def setup():
    hu={128,129}
    for n,v in _T_HELPERS.items():
        hu.add(next(a for a in range(130,_ENTRY) if a not in hu and _g(a)==v))
    ORDER=[x for x in range(130,300) if x not in hu]+list(range(300,_ENTRY))
    used=set(LABELS)|set(OTHERS)
    B={x:bfs(_g(x)) for x in ORDER}
    def cell_for(val):
        for x in ORDER:
            if x not in used and val in B[x]: used.add(x); return x
        raise RuntimeError(('no cell for',val))
    reach=set().union(*B.values())
    Gb=next(g for g in range(15000,21000) if all((g-1+k) in reach for k in range(3)))
    zcs=[cell_for(Gb-1+k) for k in range(3)]
    C0,C0v=next((x,v) for x in ORDER if x not in used for v in B[x] if rot(v)&0xFF==48); used.add(C0)
    C1,C1v=next((x,v) for x in ORDER if x not in used for v in B[x] if rot(v)&0xFF==49); used.add(C1)
    Bc=[cell_for(B) for B in BASES]
    Mc=[cell_for(MASK) for _ in range(3)]
    Rc=cell_for(Gb-1)
    # seeds: per label a walked cell + chain whose value v and f1(v) are free hub values
    taken=set(range(Gb-2,Gb+5))
    for K in KS: taken|={LANDK(K,T) for T in range(33,127)}
    seeds={}; hubvals={}
    for lab in '01xn':
        found=None
        label_cells=sorted(q for q in LABELS if LABELS[q]==lab)
        for x in ORDER:
            if x in used: continue
            for seed,path in B[x].items():
                a=seed; values=[]
                for p in label_cells:
                    a=_crazy(a,_g(p)); a=_crazy(a,a); values.append(a)
                if len(set(values))!=1: continue
                H=values[0]
                if not 10000<=H<59000 or not any(admits(H+3,e-1) for e in OTHERS): continue
                if not any(admits(H+2,p-1) for p in label_cells): continue
                cells={H,H+1,H+2,H+3}
                if cells & taken: continue
                found=(x,path,H); break
            if found: break
        x,path,H=found; used.add(x); seeds[lab]=(x,path); hubvals[lab]=(H,)
        taken|={H,H+1,H+2,H+3}
    ncells=sorted({*(T+1 for T in admissible(Gb+3) if T+1 in OTHERS),
                   *(next(e for e in OTHERS if admits(H+3,e-1))
                     for hs in hubvals.values() for H in hs)})
    found=None
    for x in ORDER:
        if x in used: continue
        for seed,path in B[x].items():
            a=seed; values=[]
            for p in ncells:
                a=_crazy(a,_g(p)); a=_crazy(a,a); values.append(a)
            if len(set(values))!=1: continue
            H=values[0]
            cells={H,H+1,H+2,H+3}
            if (10000<=H<59000 and any(admits(H+3,e-1) for e in ncells)
                    and not cells & taken):
                found=(x,path,H); break
        if found: break
    x,path,H=found; used.add(x); seeds['N']=(x,path); hubvals['N']=(H,)
    taken|={H,H+1,H+2,H+3}
    return dict(hu=hu,Gb=Gb,zcs=zcs,C0=C0,C0v=C0v,C1=C1,C1v=C1v,Bc=Bc,Mc=Mc,Rc=Rc,B=B,seeds=seeds,hubvals=hubvals,ncells=ncells,used=set(used),taken=taken)

def build(s,cf):
    Gb,zcs,Bc=cf['Gb'],cf['zcs'],cf['Bc']
    NCELLS=cf['ncells']
    used={128,129}|cf['used']; mem={a:_g(a) for a in range(_ENTRY)}
    # Each row has one state at each reachable depth.  Its parity cell is
    # consumed by ``*``, so allocate one copy per depth.
    states=[START[s]]
    if states[0] in (0,2): states.append(4)
    elif states[0]==1: states.extend((3,4))
    pcs=[]
    for depth,q in enumerate(states):
        target=PARITY[POS[q][s]%2]
        cell=next(x for x in cf['B'] if x not in used and target in cf['B'][x])
        used.add(cell); pcs.append((cell,cf['B'][cell][target]))
    helper={'all1':128,'all2':129}
    for n,v in _T_HELPERS.items():
        a=next(x for x in range(130,_ENTRY) if x not in used and _g(x)==v); used.add(a); helper[n]=a
    data={Gb+k:_char_for('o',Gb+k) for k in range(3)}
    P=_Planner(_ENTRY+1,34+(7-_ENTRY)%94,mem,data); P.code[_ENTRY]='j'
    _build_constants(P,helper)
    for c in (helper['all1'],helper['all2']): P.op('p',c); P.op('p',c)
    P.op('*',helper['w']); P.op('p',helper['all2'])
    phases=[len(P.code)]
    chains=[(f'z{k}',zcs[k],Gb-1+k) for k in range(3)]+[('C0',cf['C0'],cf['C0v']),('C1',cf['C1'],cf['C1v'])]
    chains+=[(f'B{i}',Bc[i],BASES[i]) for i in range(5)]
    chains += [(f'M{i}',cell,MASK) for i,cell in enumerate(cf['Mc'][:len(pcs)])]
    chains += [('R',cf['Rc'],Gb-1)]
    chains += [(f'P{d}',cell,PARITY[POS[states[d]][s]%2]) for d,(cell,_) in enumerate(pcs)]
    chain_cost=[]
    for name,cell,val in chains:
        before=len(P.code)
        r=bfs(_g(cell)); assert val in r,(cell,val); _emit_chain(P,cell,' '.join(r[val]),helper)
        chain_cost.append((name,len(P.code)-before))
    phases.append(len(P.code))
    # label chains: each label's cells alternate f1(v), v
    value={}
    for lab in '01xnN':
        x,path=cf['seeds'][lab]; _emit_chain(P,x,' '.join(path),helper)
        for p in (NCELLS if lab=='N' else sorted(q for q in LABELS if LABELS[q]==lab)):
            P.op('p',p); P.op('p',p); value[p]=cf['hubvals'][lab][0]
    phases.append(len(P.code))
    P.op('*',helper['all2']); P.op('p',cf['Rc'])
    P.op('*',helper['all1']); P.op('p',cf['Rc'])
    for _ in range(10):
        P.op('*',cf['Rc']); P.op('p',helper['w'])
    parity_cells=sorted(zip(pcs,states),key=lambda x:POS[x[1]][s]%2)
    for i,(mask,((cell,_),_q)) in enumerate(zip(cf['Mc'],parity_cells)):
        if i==0:
            P.op('*',helper['z0']); P.op('p',helper['w'])
        P.op('p',mask); P.op('p',cell)
    phases.append(len(P.code))
    Z0=value[NCELLS[0]]+1
    pointer={}
    for p in sorted(value):
        H=value[p]
        if admits(H+2,p-1): pointer.setdefault(H,p)
    # hub rotations: every hub value gets its own target copy
    avoid=set(cf['taken'])|set(data)
    def hub_to(Hp1,room):
        for hc in _valid_chars(Hp1):
            for k,sv in _rotations(hc):
                if 12000<=sv+1 and sv+1+room<_WORDS and not any((sv+1+o) in avoid for o in range(room)):
                    return hc,k,sv+1
        raise RuntimeError('no rotation')
    for H in set(value.values()):
        if H in pointer:
            data[H+2]=pointer[H]-1
            assert admits(H+2,pointer[H]-1)
        escape=next(e for e in NCELLS if admits(H+3,e-1))
        data[H+3]=escape-1
    copies=[]      # (key, shared entry address, room, hub turns)
    for lab in '01xn':
        key=KEY[lab]; room=140 if key[0]=='p' else 300
        hubs=sorted(set(value[p] for p in value if LABELS.get(p)==lab))
        options=[]
        for H in hubs:
            byaddr={}
            for hc in _valid_chars(H+1):
                for k,sv in _rotations(hc): byaddr.setdefault(sv+1,(hc,k))
            options.append(byaddr)
        common=set(options[0]).intersection(*options[1:])
        addr=next(a for a in sorted(common,key=lambda a:sum(o[a][1] for o in options))
                  if 7000<=a and a+room<_WORDS and not any(a+o in avoid for o in range(room)))
        turns=[]
        for H,option in zip(hubs,options):
            hc,k=option[addr]; data[H+1]=hc; turns.append(k)
            P.hub(pointer[H],H)
            for turn in range(k):
                P.raw('*')
                if turn+1<k:
                    P.raw('j'); P.d=pointer[H]
                    P.raw('j'); P.d=H+1
            P.d=H+2
        avoid|=set(range(addr,addr+room)); copies.append((key,addr,room,turns))
    phases.append(len(P.code))
    def view(q):
        c=POS[q][s]; par=(Gb+c)%2; return c,Bc[q],KS[2*q+par],par
    # landing chars: pick an admissible char whose pointer cell carries the target's label
    for q in range(5):
        c,bc,K,par=view(q); cell=Gb+c
        for T in admissible(cell):
            lab=LABOF[TGT[DELTA[q][meaning(T,par)]]]; L=LANDK(K,T)
            cL=next(ch for ch in _valid_chars(L) if LABELS.get(ch+1)==lab)
            assert data.get(L,cL)==cL,('landing clash',L); data[L]=cL
    noneset={a for a in range(_ENTRY) if P.mem.get(a) is None}
    P.mem={a:_g(a) for a in range(_ENTRY)}
    for a in noneset: P.mem[a]=None
    def decoder(Q,q,depth):
        c,bc,K,par=view(q); cell=Gb+c; z=zcs[c]
        Q.op('*',pcs[depth][0])
        Q.op('p',bc)
        Q.goto(z); Q.raw('j'); Q.d=cell; Q.raw('p'); Q.d=cell+1
        for _ in range(2-c): Q.raw('o')          # d = Gb+3, the next group's cell
        for op in 'jjoojj': Q.raw(op)
        Q.d=Z0                                   # converges whatever Gb+3 holds
        Q.goto(z); Q.raw('j'); Q.d=cell; Q.raw('j')
        Q.raw('j'); Q.raw('j'); Q.raw('i')
    decoder(P,START[s],0)
    spans=[]
    for key,addr,room,turns in copies:
        Q=_Planner(addr,0,dict(P.mem),data)
        Q.raw('o')
        Q.raw('j')
        Q.raw('j'); Q.d=Z0
        if key=='s3': decoder(Q,3,1)
        elif key=='s4': decoder(Q,4,states.index(4))
        else: Q.op('*',cf['C0'] if key=='p0' else cf['C1']); Q.raw('<'); Q.raw('v')
        assert all(addr<=a<addr+room for a in Q.code),('block overflows its room',key,addr,max(Q.code)-addr)
        spans.append((key,max(Q.code)-addr+1))
        for a,op in Q.code.items(): P.code[a]=op
    src=[_char_for('o',a) for a in range(_WORDS)]
    for a,op in P.code.items(): src[a]=_char_for(op,a)
    for a,ch in data.items(): src[a]=ch
    phases.append(len(P.code))
    return src,len(P.code),[phases[0],*(b-a for a,b in zip(phases,phases[1:]))],[k for *_,ks in copies for k in ks],spans,chain_cost

if __name__=='__main__':
    cf=setup()
    print('group',cf['Gb'],'hub values',cf['hubvals'])
    tot=bad=0
    for s in range(8):
        src,ncode,phases,turns,spans,chain_cost=build(s,cf)
        progs=[]
        for x in itertools.product(range(7),repeat=3):
            sv=list(src)
            for c in range(3):
                cell=cf['Gb']+c; par=cell%2
                sv[cell]=next(T for T in admissible(cell) if meaning(T,par)==x[c])
            nxt=cf['Gb']+3
            sv[nxt]=admissible(nxt)[(x[0]*49+x[1]*7+x[2])%8]      # neighbour table char varies
            progs.append((x,''.join(chr(v) for v in sv)))
        fails=0
        for x,prog in progs:
            Path('/private/tmp/malbolge17.mb').write_text(prog)
            out=subprocess.run(['/private/tmp/malbolge17-msim','/private/tmp/malbolge17.mb','0'],capture_output=True,text=True).stdout.split()
            got=out[1] if len(out)>1 else ''
            if got!=expected(s,x): fails+=1
        tot+=len(progs); bad+=fails
        print(f'row {s}: code cells {ncode}, phases {phases}, chain costs {chain_cost}, hub turns {turns}, spans {spans}, {len(progs)-fails}/{len(progs)} correct',flush=True)
    print('FIVE-STATE group decoder, real source:',tot-bad,'/',tot)
